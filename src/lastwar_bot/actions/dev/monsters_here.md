# What monsters the client can name around the camera, WITHOUT a config filter — #1702.
# ru: Каких монстров клиент видит вокруг камеры — без фильтра по конфигу.
#
# The chain asks `GetMonsterListInArea` with one config id (1030000) as the whitelist and
# got «none» while the operator watched dozens of golden zombies go past the camera. So
# the question is what ids are really out there: this asks with a WIDE whitelist and
# reports what came back, grouped by config id.

READ_LUA (function() local p = DataCenter.__lw_gold or {} local ws = DataCenter.__lw_gold_ws local alive = false pcall(function() alive = (ws ~= nil) and (ws.CurTilePos ~= nil) end) if not alive then ws = nil pcall(function() local arr = CS.UnityEngine.Object.FindObjectsOfType(typeof(CS.UnityEngine.MonoBehaviour)) for i = 0, arr.Length - 1 do local mb = arr[i] local n = nil pcall(function() n = mb:GetType().Name end) if n == 'WorldScene' then ws = mb break end end end) DataCenter.__lw_gold_ws = ws end if ws == nil then return 'no-worldscene' end local counts, total = {}, 0 local ok, err = pcall(function() local ids = CS.System.Collections.Generic.Dictionary(CS.System.Int32, CS.System.Int32)() for id = 1030000, 1030099 do pcall(function() ids:Add(id, 1) end) end for _, id in ipairs({1010000, 1020000, 1040000, 1050000, 1060000}) do pcall(function() ids:Add(id, 1) end) end local res = CS.System.Collections.Generic.Dictionary(CS.System.Int64, CS.UnityEngine.Vector2Int)() ws:GetMonsterListInArea(ws.CurTilePos, 2000, ids, res) local e = res:GetEnumerator() while e:MoveNext() do total = total + 1 end end) if not ok then return 'error ' .. tostring(err) end local seen1030000 = -1 pcall(function() local ids = CS.System.Collections.Generic.Dictionary(CS.System.Int32, CS.System.Int32)() ids:Add(1030000, 1) local res = CS.System.Collections.Generic.Dictionary(CS.System.Int64, CS.UnityEngine.Vector2Int)() ws:GetMonsterListInArea(ws.CurTilePos, 2000, ids, res) local n = 0 local e = res:GetEnumerator() while e:MoveNext() do n = n + 1 end seen1030000 = n end) local pos = 'nil' pcall(function() pos = tostring(ws.CurTilePos.x) .. ',' .. tostring(ws.CurTilePos.y) end) return 'wide=' .. tostring(total) .. ' only1030000=' .. tostring(seen1030000) .. ' campos=' .. pos end)() INTO monsters_here
LOG "monsters near the camera: {monsters_here}"

# …and what the SCENE itself is drawing, which is what the operator is looking at.
READ_LUA (function() local n, names = 0, {} pcall(function() local arr = CS.UnityEngine.Object.FindObjectsOfType(typeof(CS.UnityEngine.MonoBehaviour)) for i = 0, arr.Length - 1 do local mb = arr[i] local cn = nil pcall(function() cn = mb:GetType().Name end) if cn == 'TouchObjectEventTrigger' then local go = nil pcall(function() go = mb.gameObject end) local nm = nil pcall(function() nm = tostring(go.name) end) if nm ~= nil and string.find(nm, 'Monster') then n = n + 1 if #names < 6 then names[#names+1] = nm end end end end end) return 'drawn=' .. tostring(n) .. ' [' .. table.concat(names, ' ') .. ']' end)() INTO drawn
LOG "monsters drawn in the scene: {drawn}"
