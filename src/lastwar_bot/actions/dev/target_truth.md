# Ask the client about the CHOSEN tile every way it can be asked — #1702.
# ru: Спросить клиент про выбранную клетку всеми способами и сравнить ответы.
#
# «Я очищаю реестр, прохожу по миру, делаю поиск — и меня кидает на пустые координаты.»
# The confirmation we trust is `GetMonsterListInArea`, and if that answers out of a
# drawing cache it will happily confirm a corpse. So: the area list, the monster's own
# record, the tile's point info, and what the scene is drawing — side by side, for the
# very tile the button called a target.

READ_LUA (function() local p = DataCenter.__lw_gold or {} local t = p.cur if t == nil then return 'no target' end local ws = DataCenter.__lw_gold_ws local out = {'tile=' .. tostring(t.x) .. ',' .. tostring(t.y) .. ' pid=' .. tostring(t.pid) .. ' uuid=' .. tostring(t.uuid)} local inarea = '?' pcall(function() local ids = CS.System.Collections.Generic.Dictionary(CS.System.Int32, CS.System.Int32)() for _, id in ipairs(p.ids or {1030000}) do pcall(function() ids:Add(id, 1) end) end local res = CS.System.Collections.Generic.Dictionary(CS.System.Int64, CS.UnityEngine.Vector2Int)() ws:GetMonsterListInArea(CS.UnityEngine.Vector2Int(t.x, t.y), 2, ids, res) local n, hit = 0, false local e = res:GetEnumerator() while e:MoveNext() do n = n + 1 if tostring(e.Current.Key) == tostring(t.uuid) then hit = true end end inarea = 'n=' .. tostring(n) .. ' same_uuid=' .. tostring(hit) end) out[#out+1] = 'area: ' .. inarea local md = '?' pcall(function() local d = ws:GetMonsterData(t.uuid) if d == nil then md = 'nil' else local bits = {} for _, f in ipairs({'canAttack', 'hp', 'level', 'cfgId', 'state', 'dead'}) do local v = nil pcall(function() v = d[f] end) if v ~= nil then bits[#bits+1] = f .. '=' .. tostring(v) end end md = table.concat(bits, ' ') end end) out[#out+1] = 'GetMonsterData: ' .. md local pi = '?' pcall(function() local pm = ws.PointManager local has = ws:HasPointInfo(t.pid) if not has then pi = 'not-loaded' else local info = pm:GetPointInfo(t.pid) if info == nil then pi = 'nil' else local bits = {} for _, f in ipairs({'pointType', 'uuid', 'monsterUuid', 'cfgId'}) do local v = nil pcall(function() v = info[f] end) if v ~= nil then bits[#bits+1] = f .. '=' .. tostring(v) end end pi = table.concat(bits, ' ') end end end) out[#out+1] = 'point: ' .. pi return table.concat(out, ' || ') end)() INTO truth
