# Probe: which fields on the client's own character carry a uid and a face (#2061).
# ru: Проба: какие поля своего персонажа несут uid и картинку.
#
# The header is to draw the player's OWN avatar, and the face plumbing
# (`tools/lib/player_faces.py`) wants a uid and a head-icon id. `LuaEntry.Player` is the
# object the profile card is already read from (`actions/read_player_profile.md`), so the
# question is only WHICH of its fields answer. Names are probed, never assumed.
#
# Nothing is pressed, nothing is sent and no window is opened.
READ_LUA (function() local P = LuaEntry and LuaEntry.Player if P == nil then return 'no player' end local out = {} local function put(name, value) if value ~= nil and value ~= '' and value ~= 0 then out[#out+1] = name .. '=' .. tostring(value) end end for _, key in ipairs({'headPic','pic','picture','figure','frame','frameId','headFrame','iconId','photo','face','avatarId','headid'}) do local ok, v = pcall(function() return P[key] end) if ok then put('.' .. key, v) end end for _, name in ipairs({'GetHeadPic','GetPic','GetFace','GetPhoto','GetFigure','GetHeadFrame','GetIconId','GetHeadUrl','GetPicUrl'}) do local ok, v = pcall(function() return P[name] and P[name](P) end) if ok then put(name .. '()', v) end end return table.concat(out, ' | ') end)() INTO me
LOG "own card fields: {me}"
